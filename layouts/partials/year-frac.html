{{- /* ISO date ([-]YYYY, [-]YYYY-MM or [-]YYYY-MM-DD) to an astronomical decimal year for chart positions. */ -}} {{- $s := strings.TrimSpace (string .) -}} {{- $y := float (partial "iso-year.html" $s) -}} {{- $p := split (strings.TrimPrefix "-" $s) "-" -}} {{- if gt (len $p) 1 -}} {{- $m := float (strings.TrimLeft "0" (index $p 1) | default "1") -}} {{- $y = add $y (div (sub $m 1.0) 12.0) -}} {{- end -}} {{- if gt (len $p) 2 -}} {{- $d := float (strings.TrimLeft "0" (index $p 2) | default "0") -}} {{- $y = add $y (div $d 366.0) -}} {{- end -}} {{- return $y -}}
