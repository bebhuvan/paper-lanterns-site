{{- /* Year by year: one small block per dated letter, stacked by year, with each year's count above its stack. The writer's life periods, when the book has them, are named above with hairline edges; periods marked in data/marked_periods.json are shaded across the full height and their names darkened. The longest silence of three or more years is named and bracketed. If the home page's P.S. letter is in this book, its block is lit and a leader line runs up to a short note. Context: the book page. Draws in a 1000-unit-wide viewBox; the chart box scrolls sideways on phones. */ -}} {{- $yc := .Params.year_counts -}} {{- $periods := .Params.periods | default slice -}} {{- $mark := index site.Data.marked_periods .Params.book_id -}} {{- $marked := slice -}} {{- with $mark -}} {{- $marked = .periods -}} {{- $labels := slice }}{{ range $periods }}{{ $labels = $labels | append .label }}{{ end -}} {{- range $marked }}{{ if not (in $labels .) }}{{ errorf "data/marked_periods.json: %q is not a period of %s" . $.Params.book_id }}{{ end }}{{ end -}} {{- end -}} {{- $W := 1000.0 -}} {{- $first := int (index (index $yc 0) 0) -}} {{- $last := int (index (index $yc (sub (len $yc) 1)) 0) -}} {{- $lo := float $first -}}{{- $hi := add (float $last) 1 -}} {{- range $periods -}} {{- $s := partial "year-frac.html" .start -}}{{- $e := partial "year-frac.html" .end -}} {{- if lt $s $lo }}{{ $lo = $s }}{{ end }}{{ if gt $e $hi }}{{ $hi = $e }}{{ end -}} {{- end -}} {{- $lo = sub (math.Floor $lo) 1 -}}{{- $hi = add (math.Ceil $hi) 1 -}} {{- $scale := div $W (sub $hi $lo) -}} {{- $maxN := 0 -}}{{- $peakYear := 0 -}} {{- range $yc }}{{ if gt (int (index . 1)) $maxN }}{{ $maxN = int (index . 1) }}{{ $peakYear = int (index . 0) }}{{ end }}{{ end -}} {{- /* Blocks are 14 by 7 with a 3 gap; long spans narrow them and very busy years shorten them so the tallest stack stays within 170. */ -}} {{- $bw := math.Min 14 (mul $scale 0.72) -}} {{- $unit := 10.0 -}}{{- if gt (mul $maxN 10) 170 }}{{ $unit = div 170.0 $maxN }}{{ end -}} {{- $bh := mul $unit 0.7 -}} {{- $showCounts := ge $scale 16 -}} {{- /* The featured letter, if it belongs to this book. */ -}} {{- $feat := site.GetPage site.Home.Params.featured_letter -}} {{- $featYear := 0 -}}{{- $featIdx := -1 -}}{{- $featN := 0 -}} {{- if and $feat (eq $feat.Parent.RelPermalink .RelPermalink) $feat.Params.letter_date.iso -}} {{- $featYear = int (index (split $feat.Params.letter_date.iso "-") 0) -}} {{- $same := slice -}} {{- range sort (where (where site.RegularPages "Params.book_id" .Params.book_id) "Params.letter_date.iso" "!=" "") "Params.letter_date.iso" -}} {{- if eq (int (index (split .Params.letter_date.iso "-") 0)) $featYear }}{{ $same = $same | append . }}{{ end -}} {{- end -}} {{- range $i, $l := $same }}{{ if eq $l.RelPermalink $feat.RelPermalink }}{{ $featIdx = $i }}{{ end }}{{ end -}} {{- end -}} {{- /* Period names go in up to three rows; a name drops a row when it would run into the one before it. */ -}} {{- $rows := slice -}}{{- $rowEnd := slice -1000.0 -1000.0 -1000.0 -}}{{- $rowsUsed := 0 -}} {{- range $periods -}} {{- $x0 := mul (sub (partial "year-frac.html" .start) $lo) $scale -}} {{- $w := mul (len .label) 9.6 -}} {{- $r := 2 -}} {{- range $k := seq 2 0 }}{{ if gt (add $x0 2) (index $rowEnd $k) }}{{ $r = $k }}{{ end }}{{ end -}} {{- $newEnd := slice -}}{{- range $k, $v := $rowEnd }}{{ $newEnd = $newEnd | append (cond (eq $k $r) (add (add $x0 $w) 18) $v) }}{{ end }}{{ $rowEnd = $newEnd -}} {{- $rows = $rows | append $r -}}{{- if ge $r $rowsUsed }}{{ $rowsUsed = add $r 1 }}{{ end -}} {{- end -}} {{- $T := 0.0 -}}{{- if $periods }}{{ $T = add (mul $rowsUsed 28) 10.0 }}{{ end -}} {{- $blocksH := mul $maxN $unit -}} {{- $Hc := math.Max 200 (add $blocksH 122) -}} {{- $base := add $T $Hc -}} {{- $room := sub $Hc (add $blocksH 22) -}} {{- /* The longest run of years with no letter. */ -}} {{- $prev := 0 -}}{{- $gapLen := 0 -}}{{- $gapAt := 0 -}} {{- range $yc -}} {{- $y := int (index . 0) -}} {{- if and $prev (gt (sub (sub $y $prev) 1) $gapLen) }}{{ $gapLen = sub (sub $y $prev) 1 }}{{ $gapAt = add $prev 1 }}{{ end -}} {{- $prev = $y -}} {{- end -}} {{- $words := dict "3" "three" "4" "four" "5" "five" "6" "six" "7" "seven" "8" "eight" "9" "nine" "10" "ten" -}} {{- $gapWord := index $words (string $gapLen) | default (string $gapLen) -}} {{- $span := sub $hi $lo -}} {{- $every := 5 -}}{{- if gt $span 60 }}{{ $every = 10 }}{{ end }}{{ if gt $span 150 }}{{ $every = 50 }}{{ end }}{{ if gt $span 500 }}{{ $every = 100 }}{{ end -}} {{- $H := add $base 40 -}} {{- $label := printf "%d dated letters by year, %d to %d, each letter a small block" (int .Params.counts.dated_letters) $first $last -}} {{- with $periods }}{{ $label = printf "%s, set against the periods of the writer’s life" $label }}{{ end -}} {{- if ge $gapLen 3 }}{{ $label = printf "%s. None survive from %d to %d" $label $gapAt (add $gapAt (sub $gapLen 1)) }}{{ end -}} {{- $label = printf "%s. The busiest year is %d, with %d %s." $label $peakYear $maxN (cond (eq $maxN 1) "letter" "letters") -}}
{{- $g0 := 0.0 -}}{{- $g1 := 0.0 -}}{{- $gm := 0.0 -}}{{- $gy := 0.0 -}}{{- $gx := 0.0 -}}{{- $gL := 0.0 -}}{{- $gR := 0.0 -}}{{- $gLines := slice (printf "%s years" $gapWord) "without a letter" -}} {{- if ge $gapLen 3 -}} {{- $g0 = mul (sub (float $gapAt) $lo) $scale -}}{{- $g1 = mul (sub (float (add $gapAt $gapLen)) $lo) $scale -}} {{- $gm = div (add $g0 $g1) 2 -}}{{- $gy = add $T (math.Max 30 (mul $room 0.42)) -}} {{- /* Always centred on the silence; when wider than the gap it overhangs evenly, and band edges step around it. */ -}} {{- /* Two lines need about 118 units; a narrower silence gets three shorter lines (about 76). */ -}} {{- $tw := 118.0 -}}{{- if gt $tw (sub $g1 $g0) }}{{ $tw = 76.0 }}{{ $gLines = slice (printf "%s years" $gapWord) "without" "a letter" }}{{ end -}} {{- $gx = $gm -}}{{- $gL = sub $gm (div $tw 2) -}}{{- $gR = add $gL $tw -}} {{- end -}}
