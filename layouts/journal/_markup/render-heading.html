", "Part 3: ") is left out rather than shown twice. */ -}} {{- if eq .Level 1 }}{{ errorf "%s: the heading %q uses a single #; the title comes from front matter, so start sections with ## (and sub-sections with ###)" .Page.File.Path .PlainText }}{{ end -}} {{- $text := .Text -}} {{- if eq .Level 2 }}{{ $text = replaceRE `^(?:(?:Part|Section)\s+)?(?:[IVXLC]+|[0-9]+)[.:)]\s+` "" $text }}{{ end -}} {{ $text | safeHTML }}
